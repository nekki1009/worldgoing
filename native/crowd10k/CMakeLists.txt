cmake_minimum_required(VERSION 3.16)
project(crowd10k CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if (MSVC)
  add_compile_options(/O2 /fp:precise /W4)
else()
  add_compile_options(-O2 -Wall -Wextra)
endif()
find_package(Threads REQUIRED)
add_library(crowd_core STATIC src/crowd_core.cpp src/combat.cpp src/command.cpp src/ai.cpp src/crowd_c_api.cpp src/state.cpp src/render.cpp)
target_link_libraries(crowd_core PUBLIC Threads::Threads)
add_executable(crowd_bench src/bench.cpp)
target_link_libraries(crowd_bench crowd_core)
add_executable(crowd_api_test src/api_test.c)
target_link_libraries(crowd_api_test crowd_core)
add_executable(crowd_rules_vectors tests/rules_vectors.cpp)
add_executable(crowd_scenario_tests tests/scenario_tests.cpp)
target_link_libraries(crowd_scenario_tests crowd_core)
add_executable(crowd_fire_host_bridge_test tests/fire_host_bridge_test.cpp)
target_link_libraries(crowd_fire_host_bridge_test crowd_core)
add_executable(crowd_flicker tools/flicker.cpp)
target_link_libraries(crowd_flicker crowd_core)
